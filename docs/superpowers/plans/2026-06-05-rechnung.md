```ts
import type { PrismaClient } from '../../prisma/generated/client.js';

export async function generiereRechnungsNummer(
  prisma: PrismaClient,
): Promise<{ nummer: string; jahr: number; laufendeNummer: number }> {
  const jahr = new Date().getFullYear();

  const letzteRechnung = await prisma.rechnung.findFirst({
    where: { jahr },
    orderBy: { laufendeNummer: 'desc' },
  });

  const laufendeNummer = letzteRechnung ? letzteRechnung.laufendeNummer + 1 : 1;
  const nummer = `${laufendeNummer} / ${jahr}`;

  return { nummer, jahr, laufendeNummer };
}
```

```ts
export function validiereRechnungsVorlage(buffer: Buffer): VorlagenValidierungsfehler | null {
  const zip = new PizZip(buffer);
  const docXml = zip.files['word/document.xml']?.asText() ?? '';

  const fehlend = PFLICHTFELDER_RECHNUNG.filter((p) => !docXml.includes(p));

  if (fehlend.length > 0) {
    return { typ: 'RECHNUNG', fehlendePflichtfelder: fehlend };
  }
  return null;
}
```

```ts
export async function generiereRechnungPdf(
  prisma: PrismaClient,
  revisionId: number,
): Promise<string> {
  const revision = await prisma.rechnungRevision.findUniqueOrThrow({
    where: { id: revisionId },
    include: {
      rechnung: true,
      vorlage: true,
    },
  });

  const positionen: Array<{ text: string; priceInCents: number }> = JSON.parse(revision.positionen);
  const netto = positionen.reduce((sum, p) => sum + p.priceInCents, 0);
  const steuer = Math.round(netto * (revision.ust / 100));
  const brutto = netto + steuer;

  const todayFormatted = new Date().toLocaleDateString('de');
  const invoiceDateFormatted = revision.rechnungsdatum.toLocaleDateString('de');

  const paymentDetails = revision.einzug
    ? `Der Betrag wird in den nächsten Tagen mit dem Vermerk "${revision.rechnung.nummer}" von Ihrem Konto eingezogen.`
    : 'Bitte Überweisen Sie den Betrag unter Angabe der Rechnungsnummer an das untenstehende Konto.';

  const tableXml = getDocxTablePositionen(positionen);

  const vorlageBuffer = await fs.promises.readFile(toAbsolutePath(revision.vorlage.dateiPfad));

  // Pass 1: Template-level placeholders
  const zip1 = new PizZip(vorlageBuffer);
  const doc1 = new Docxtemplater(zip1, {
    paragraphLoop: true,
    linebreaks: true,
  });
  doc1.render({
    Anrede: revision.anrede,
    Name: revision.name,
    Strasse: revision.strasse,
    Ort: revision.ort,
    Heute: invoiceDateFormatted,
    Title: revision.title,
    ReNr: revision.rechnung.nummer,
    Netto: (netto / 100).toFixed(2),
    Steuer: (steuer / 100).toFixed(2),
    Brutto: (brutto / 100).toFixed(2),
    ust: revision.ust,
    MailAnrede: revision.mailAnrede,
    PaymentDetails: paymentDetails,
    TableWithServices: tableXml,
    monthlyPay: false,
    MonthlyBruto: '',
  });

  // Pass 2: Per-position placeholders
  const doc2 = new Docxtemplater(doc1.getZip(), {
    paragraphLoop: true,
    linebreaks: true,
  });
  const pass2Data: Record<string, string> = {};
  positionen.forEach((pos, idx) => {
    pass2Data[`text${idx}`] = pos.text;
    pass2Data[`price${idx}`] = (pos.priceInCents / 100).toFixed(2);
  });
  doc2.render(pass2Data);

  const docxBuffer = doc2.getZip().generate({ type: 'nodebuffer', compression: 'DEFLATE' }) as Buffer;
  const pdfBuffer = await docxZuPdf(docxBuffer);

  const outputDir = path.join(
    RECHNUNGEN_DIR,
    String(revision.rechnungId),
    `r${revision.revisionNummer}`,
  );
  await fs.promises.mkdir(outputDir, { recursive: true });

  const dateiName = `${revision.rechnung.nummer.replace(/\s*\/\s*/g, '-')}-R${revision.revisionNummer}.pdf`;
  const outputPath = path.join(outputDir, dateiName);
  await fs.promises.writeFile(outputPath, pdfBuffer);

  return toRelativePath(outputPath);
}
```

```ts
function revisionZuDto(rev: {
  id: number;
  revisionNummer: number;
  status: string;
  pdfStatus: string;
  pdfFehler: string | null;
  pdfTechFehler: string | null;

  };

function validierePflichtfelder(eingaben: RechnungEingaben): string | null {
  if (!eingaben.name.trim()) return 'Name ist erforderlich';
  if (!eingaben.strasse.trim()) return 'Straße ist erforderlich';
  if (!eingaben.ort.trim()) return 'Ort ist erforderlich';
  if (!eingaben.title.trim()) return 'Titel ist erforderlich';
  if (!eingaben.positionen || eingaben.positionen.length === 0) {
    return 'Mindestens eine Position ist erforderlich';
  }
  for (let i = 0; i < eingaben.positionen.length; i++) {
    if (!eingaben.positionen[i].text.trim()) {
      return `Position ${i + 1}: Text ist erforderlich`;
    }
    if (eingaben.positionen[i].priceInCents <= 0) {
      return `Position ${i + 1}: Preis muss größer als 0 sein`;
    }
  }
  if (eingaben.ust !== 7 && eingaben.ust !== 19) {
    return 'USt muss 7% oder 19% sein';
  }
  return null;
}

function berechneBrutto(positionenJson: string, ust: number): number {
  const positionen: Rechnungsposition[] = JSON.parse(positionenJson);
  const netto = positionen.reduce((sum, p) => sum + p.priceInCents, 0);
  const steuer = Math.round(netto * (ust / 100));
  return netto + steuer;
}

async function updateRevisionPdfStatus(
  revisionId: number,
  update: {
    pdfStatus: PdfStatus;
    dateiPfad?: string | null;
    pdfFehler?: string | null;
    pdfTechFehler?: string | null;
  },
) {
  await prisma.rechnungRevision.update({
    where: { id: revisionId },
    data: {
      pdfStatus: update.pdfStatus,
      ...(update.dateiPfad !== undefined && { dateiPfad: update.dateiPfad }),
      ...(update.pdfFehler !== undefined && { pdfFehler: update.pdfFehler }),
      ...(update.pdfTechFehler !== undefined && { pdfTechFehler: update.pdfTechFehler }),
    },
  });
}

function serialisiereTechnischenFehler(error: unknown): string {
  if (error instanceof Error) {
    return error.stack || `${error.name}: ${error.message}`;
  }
  return String(error);
}

async function versuchePdfErzeugung(revisionId: number) {
  try {
    const dateiPfad = await generiereRechnungPdf(prisma, revisionId);
    await updateRevisionPdfStatus(revisionId, {
      pdfStatus: 'OK',
      dateiPfad,
      pdfFehler: null,
      pdfTechFehler: null,
    });
    return {
      pdfStatus: 'OK' as const,
      pdfFehler: null,
      pdfTechFehler: null,
      dateiPfad,
    };
  } catch (error) {
    await updateRevisionPdfStatus(revisionId, {
      pdfStatus: 'FAILED',
      dateiPfad: null,
      pdfFehler: PDF_FEHLER_MELDUNG,
      pdfTechFehler: serialisiereTechnischenFehler(error),
    });

    const failed = await prisma.rechnungRevision.findUniqueOrThrow({
      where: { id: revisionId },
      select: { pdfFehler: true, pdfTechFehler: true },
    });

    return {
      pdfStatus: 'FAILED' as const,
      pdfFehler: failed.pdfFehler,
      pdfTechFehler: failed.pdfTechFehler,
      dateiPfad: null,
    };
  }
}
```

```vue
  clearTimeout(debounceTimer);
  if (!empfaengerSuche.value.trim()) {
    empfaengerVorschlaege.value = [];
    return;
  }
  debounceTimer = setTimeout(async () => {
    empfaengerLaden.value = true;

      empfaengerLaden.value = false;

  empfaenger.value = e;
  neuerEmpfaenger.value = false;
  empfaengerSuche.value = e.name;
  empfaengerVorschlaege.value = [];
  eingaben.value.name = e.name;
  eingaben.value.strasse = e.strasse;
  eingaben.value.ort = e.ort;

  eingaben.value.anrede = e.anrede ?? '';

  neuerEmpfaenger.value = true;
  empfaenger.value = null;
  empfaengerVorschlaege.value = [];

  eingaben.value.positionen.push({ text: '', priceInCents: 0 });

function positionEntfernen(index: number) {
  if (eingaben.value.positionen.length > 1) {
    eingaben.value.positionen.splice(index, 1);
  }

function preisZuEuro(index: number): string {
  const cents = eingaben.value.positionen[index].priceInCents;
  return cents ? (cents / 100).toFixed(2).replace('.', ',') : '';
}

function preisVonEuro(index: number, value: string) {
  const cleaned = value.replace(',', '.').replace(/[^0-9.]/g, '');
  eingaben.value.positionen[index].priceInCents = Math.round(parseFloat(cleaned || '0') * 100);
}

const brutto = ref('0,00');
function updateBrutto() {
  const netto = eingaben.value.positionen.reduce((sum, p) => sum + p.priceInCents, 0);
  const steuer = Math.round(netto * (eingaben.value.ust / 100));
  const total = (netto + steuer) / 100;
  brutto.value = total.toFixed(2).replace('.', ',');
}

    if (istNeueRevision) {
```

```vue
import type { PdfStatus, RechnungDto, RechnungRevisionDto } from 'shared';
import { onMounted, ref } from 'vue';
import { useRoute, useRouter } from 'vue-router';

const route = useRoute();
const router = useRouter();
const id = parseInt(String(route.params.id));

const rechnung = ref<RechnungDto | null>(null);
const laden = ref(false);
const fehler = ref('');
const storniertFehler = ref('');
const retryPdfErfolg = ref<Record<number, string>>({});
const retryPdfFehler = ref<Record<number, string>>({});
const retryLaden = ref<Record<number, boolean>>({});

async function laden_() {
  laden.value = true;
  try {
    const res = await fetch(`/api/rechnung/${id}`);
    if (!res.ok) {
      fehler.value = 'Rechnung konnte nicht geladen werden';
      return;
    }
    rechnung.value = await res.json();
  } catch {
    fehler.value = 'Netzwerkfehler';
  } finally {
    laden.value = false;
  }
}

onMounted(laden_);

function formatBetrag(cents: number) {
  return new Intl.NumberFormat('de-DE', { style: 'currency', currency: 'EUR' }).format(cents / 100);
}

function formatDatum(iso: string) {
  return new Intl.DateTimeFormat('de-DE').format(new Date(iso));
}

function berechneBrutto(rev: RechnungRevisionDto): number {
  const netto = rev.eingaben.positionen.reduce((sum, p) => sum + p.priceInCents, 0);
  const steuer = Math.round(netto * (rev.eingaben.ust / 100));
  return netto + steuer;
}

function ustLabel(ust: number): string {
  return ust === 7 ? '7%' : '19%';
}

function zahlungsartLabel(einzug: boolean): string {
  return einzug ? 'Lastschrift (Einzug)' : 'Überweisung';
}

async function stornieren() {
  storniertFehler.value = '';
  try {
    const res = await fetch(`/api/rechnung/${id}/stornieren`, { method: 'PATCH' });
    if (!res.ok) {
      const err = await res.json();
      storniertFehler.value = err.error || 'Fehler beim Stornieren';
      return;
    }
    await laden_();
  } catch {
    storniertFehler.value = 'Netzwerkfehler';
  }
}

async function retryPdf(revisionId: number) {
  retryPdfErfolg.value[revisionId] = '';
  retryPdfFehler.value[revisionId] = '';
  retryLaden.value[revisionId] = true;

  try {
    const res = await fetch(`/api/rechnung/revision/${revisionId}/retry-pdf`, { method: 'POST' });
    const result = await res.json();

    if (res.ok) {
      retryPdfErfolg.value[revisionId] = 'PDF erfolgreich generiert';
      await laden_();
    } else {
      retryPdfFehler.value[revisionId] = result.error || 'Fehler beim Retry';
    }
  } catch {
    retryPdfFehler.value[revisionId] = 'Netzwerkfehler';
  } finally {
    retryLaden.value[revisionId] = false;
  }
}

function downloadPdf(revisionId: number) {
  window.open(`/api/rechnung/revision/${revisionId}/download`, '_blank');
}

function pdfStatusBadge(status: PdfStatus) {
  const map = {
    PENDING: 'bg-gray-100 text-gray-600',
    OK: 'bg-green-100 text-green-700',
    FAILED: 'bg-red-100 text-red-700',
  };
  return `inline-block text-xs font-medium px-2 py-0.5 rounded-full ${map[status] || ''}`;
}
```
